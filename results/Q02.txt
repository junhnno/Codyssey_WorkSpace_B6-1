-- Q02: 가격이 가장 비싼 도서 TOP 5
-- 기대: 5행, 38000원 '모던 C++ 프로그래밍'부터 내림차순
SELECT title, price
FROM book
ORDER BY price DESC
LIMIT 5;

-- 실행 결과 --
+------------------------+-------+
| title                  | price |
+------------------------+-------+
| 모던 C++ 프로그래밍    | 38000 |
| 파이썬 알고리즘 인터뷰 | 36000 |
| 데이터베이스 설계 기초 | 30000 |
| 우주의 구조            | 27000 |
| 파이썬 입문            | 25000 |
+------------------------+-------+
(5 rows)
