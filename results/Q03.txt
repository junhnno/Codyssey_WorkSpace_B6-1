-- Q03: 제목에 '파이썬'이 들어간 도서 검색
-- 기대: 2행 ('파이썬 입문', '파이썬 알고리즘 인터뷰')
SELECT id, title, author
FROM book
WHERE title LIKE '%파이썬%';

-- 실행 결과 --
+----+------------------------+--------+
| id | title                  | author |
+----+------------------------+--------+
| 1  | 파이썬 입문            | 홍길동 |
| 3  | 파이썬 알고리즘 인터뷰 | 박상길 |
+----+------------------------+--------+
(2 rows)
