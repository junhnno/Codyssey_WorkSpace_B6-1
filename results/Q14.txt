-- Q14: 대여 기록이 없는 회원(임하은, id=10) 탈퇴 처리
-- 기대: 1행 삭제, 회원 수 10 → 9
-- (대여 기록이 있는 회원을 지우려 하면 FK 때문에 실패한다)

WHERE id = 10;
-- 확인
SELECT COUNT(*) AS member_count FROM member;

(변경된 행 수: 1)

+--------------+
| member_count |
+--------------+
| 9            |
+--------------+
