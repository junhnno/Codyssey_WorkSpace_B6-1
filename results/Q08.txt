-- Q08: [LEFT JOIN] 한 번도 대여되지 않은 도서
-- 기대: 2행 ('조용한 아침', '습관의 기술') — 대여 기록과 매칭되지 않아 r.id가 NULL인 도서
SELECT b.id, b.title
FROM book b
LEFT JOIN rental r ON r.book_id = b.id
WHERE r.id IS NULL;

-- 실행 결과 --
+----+-------------+
| id | title       |
+----+-------------+
| 8  | 조용한 아침 |
| 12 | 습관의 기술 |
+----+-------------+
(2 rows)
